main_sensor_dht.ino 
#include <DHT.h>
#include <LiquidCrystal_I2C.h>

#define DHTPIN 2
#define DHTTYPE DHT11

DHT dht(DHTPIN, DHTTYPE);
LiquidCrystal_I2C lcd(0x27, 16, 2);

void setup() {
  lcd.init();
  lcd.backlight();
  dht.begin();
}

void loop() {
  float temp = dht.readTemperature();
  float humid = dht.readHumidity();
  
  lcd.setCursor(0,0);
  lcd.print("Temp: "); lcd.print(temp);
  delay(2000);
}

Inspektor Standar Struktur Folder & Manajemen Aset (Wajib)

Struktur folder wajib portofolio-namasiswa sesuai standar industri:

portofolio-namasiswa/
|
|-- index.html              <-- Halaman Utama (Beranda / Profil Ringkas)
|
|-- assets/                 <-- Tempat semua berkas pendukung
|   |-- css/
|   |   `-- style.css       <-- File External CSS Utama
|   |-- js/
|   |   `-- script.js       <-- File JavaScript (opsional / untuk tahap lanjut)
|   |-- img/                <-- Semua gambar & ikon
|   |   |-- profil.jpg      <-- Foto diri formal/semi-formal
|   |   |-- banner.jpg      <-- Banner header
|   |   `-- projects/       <-- Foto dokumentasi karya & tugas
|   `-- docs/               <-- Tempat file download (CV/Laporan PDF jika ada)
|
`-- pages/                  <-- Halaman modular masa depan
    |-- tentang.html        <-- Biodata & Profil Lengkap
    |-- jaringan.html       <-- Portofolio Bab Jaringan Komputer & Topologi
    |-- iot-arduino.html    <-- Portofolio Bab IoT & Simulasi Arduino
    |-- karya-proyek.html   <-- Galeri Proyek Digital & Portofolio Karya
    `-- kontak.html         <-- Form Kontak & Media Sosial
Aturan Standar Aset & Relatif Linking Path:
  • External CSS Wajib: Semua styling 100% wajib menggunakan External CSS (assets/css/style.css). Dilarang memakai Inline CSS.
  • Penamaan File/Folder: Gunakan huruf kecil dan tanda hubung tanpa spasi (contoh: iot-arduino.html, profil.jpg).
  • Di index.html: CSS diakses via <link rel="stylesheet" href="assets/css/style.css">
  • Di folder pages/*.html: CSS diakses via <link rel="stylesheet" href="../assets/css/style.css">
  • Tautan kembali ke beranda dari pages/: <a href="../index.html">Beranda</a>

Langkah Publikasi (Deployment GitHub Pages / Netlify)

  1. 1. Verifikasi Lokal Pastikan seluruh file HTML (index.html, pages/*.html) dan CSS (assets/css/style.css) telah diverifikasi tanpa broken link.
  2. 2. Inisialisasi Git & Commit Jalankan perintah Git di terminal folder proyek: git init
    git add .
    git commit -m "Initial commit portofolio XII"
  3. 3. Push ke GitHub Repository Buat repository publik bernama portofolio-namasiswa di GitHub, lalu hubungkan remote repo dan push branch main.
  4. 4. Aktifkan GitHub Pages Buka Settings > Pages > pilih Source Branch: main / root. Klik Save. Website aktif dalam 1-2 menit!
